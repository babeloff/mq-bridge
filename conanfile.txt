# The native C library the `link-dynamic` build variant needs, for development.
#
# `just build-dynamic` (`--features full-dynamic`) links librdkafka from the
# environment instead of compiling it from rdkafka-sys' bundled tree, and finds
# it through pkg-config. That is what a conda-forge or distro package does;
# this file is how a developer gets the same thing on a machine without it.
#
# `just conan` installs it and writes rdkafka.pc; `just build-dynamic` then
# puts that on PKG_CONFIG_PATH. Nothing else in the repository depends on
# Conan — the default `full` build compiles its own copy and needs none of
# this.
#
# librdkafka only. SQLite used to be here too, because `link-dynamic` selected
# sqlx's `sqlite-unbundled`; it no longer does — SQLite is bundled either way,
# so there is nothing to provision, and with `sqlite-unbundled` went the only
# bindgen in the tree, so there is no libclang to provide either.
#
# NOT what the packages link. conda-forge builds against conda-forge's
# librdkafka, a distro against the distro's. This only has to satisfy the same
# pkg-config probe locally.

[requires]
# The exact version rdkafka-sys 4.10 probes for (`rdkafka >= 2.12.1`), and the
# same one it would have compiled from its bundled tree — so the headers the
# bindings are generated against match what the static variant uses.
librdkafka/2.12.1

[options]
# Shared, or there is nothing for `link-dynamic` to link against and the whole
# point is lost.
librdkafka/*:shared=True
# librdkafka's own optional codecs and TLS. Kept on because a route that talks
# to a real broker generally needs them, and because the static variant gets
# gzip via `rdkafka/libz-static` — leaving zlib off here would make the two
# variants differ in capability rather than only in provenance.
librdkafka/*:with_ssl=True
librdkafka/*:with_zlib=True
librdkafka/*:with_zstd=True

[generators]
# Writes rdkafka.pc, which is the only interface rdkafka-sys uses to find this.
PkgConfigDeps
